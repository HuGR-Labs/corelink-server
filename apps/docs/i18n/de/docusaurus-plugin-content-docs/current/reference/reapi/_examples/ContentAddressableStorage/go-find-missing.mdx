# Go — `FindMissingBlobs`

## Prerequisites

- `go get go.corelink.humangr.com/corelink-go@latest`
- `CORELINK_PAT` exported

## Code

```go
package main

import (
    "context"
    "crypto/sha256"
    "encoding/hex"
    "fmt"
    "os"

    corelink "go.corelink.humangr.com/corelink-go"
)

func digest(b []byte) corelink.Digest {
    s := sha256.Sum256(b)
    return corelink.Digest{Hash: hex.EncodeToString(s[:]), SizeBytes: int64(len(b))}
}

func main() {
    ctx := context.Background()
    client, _ := corelink.NewClient(ctx, corelink.Config{
        Endpoint: "https://corelink-api.humangr.com",
        PAT:      os.Getenv("CORELINK_PAT"),
    })
    defer client.Close()

    blobs := [][]byte{[]byte("alpha"), []byte("beta"), []byte("gamma")}
    digests := make([]corelink.Digest, len(blobs))
    for i, b := range blobs {
        digests[i] = digest(b)
    }
    missing, err := client.CAS.FindMissingBlobs(ctx, digests)
    if err != nil {
        panic(err)
    }
    fmt.Printf("%d/%d missing\n", len(missing), len(digests))
}
```

## Expected output

```text
2/3 missing
```
