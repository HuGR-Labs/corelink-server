# Node / TypeScript — guard on server REAPI version

Refuse to proceed if the server's advertised REAPI version range does not include the client's compiled version.

## Code

```ts
import { CoreLinkClient } from "@corelink/client";

const CLIENT_REAPI = { major: 2, minor: 12, patch: 0 };

const client = new CoreLinkClient({
  endpoint: "https://acme-dev.corelink.humangr.com",
  pat: process.env.CORELINK_PAT!,
});

const caps = await client.capabilities.get();
const lo = caps.lowApiVersion;
const hi = caps.highApiVersion;
const ok =
  semverGte(CLIENT_REAPI, lo) && semverGte(hi, CLIENT_REAPI);
if (!ok) {
  throw new Error(
    `server supports ${fmt(lo)}–${fmt(hi)}, client is ${fmt(CLIENT_REAPI)}`,
  );
}
console.log("version handshake OK");

function semverGte(a: { major: number; minor: number; patch: number }, b: typeof a): boolean {
  return (
    a.major > b.major ||
    (a.major === b.major && (a.minor > b.minor || (a.minor === b.minor && a.patch >= b.patch)))
  );
}
function fmt(v: { major: number; minor: number; patch: number }): string {
  return `${v.major}.${v.minor}.${v.patch}`;
}
```

## Expected output

```text
version handshake OK
```

## Common errors

| Failure | Cause | Fix |
| --- | --- | --- |
| Thrown error with version range | Client SDK older / newer than server | Upgrade SDK or pin server version. |
