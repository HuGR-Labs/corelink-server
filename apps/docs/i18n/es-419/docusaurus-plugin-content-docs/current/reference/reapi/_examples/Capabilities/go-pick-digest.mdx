# Go — pick a digest function

Negotiate SHA256 vs BLAKE3 based on server support.

## Code

```go
package main

import (
    "context"
    "fmt"
    "os"
    "slices"

    corelink "go.corelink.humangr.com/corelink-go"
)

func main() {
    ctx := context.Background()
    client, _ := corelink.NewClient(ctx, corelink.Config{
        Endpoint: "https://corelink-api.humangr.com",
        PAT:      os.Getenv("CORELINK_PAT"),
    })
    defer client.Close()

    caps, err := client.Capabilities.Get(ctx)
    if err != nil {
        panic(err)
    }
    df := corelink.DigestSHA256
    if slices.Contains(caps.CacheCapabilities.DigestFunctions, corelink.DigestBLAKE3) {
        df = corelink.DigestBLAKE3
    }
    client.SetDigestFunction(df)
    fmt.Println("using", df)
}
```

## Expected output

```text
using BLAKE3
```
